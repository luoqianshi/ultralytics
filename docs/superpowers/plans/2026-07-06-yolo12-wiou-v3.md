```python
def wiou_v3(
    box1: torch.Tensor,
    box2: torch.Tensor,
    iou_mean: torch.Tensor,
    xywh: bool = False,
    alpha: float = 1.7,
    delta: float = 2.7,
    eps: float = 1e-7,
) -> torch.Tensor:
    """Calculate Wise-IoU v3 with dynamic non-monotonic focusing mechanism.

    Implements WIoU v3 from https://github.com/Instinct323/Wise-IoU. Returns IoU similarity
    ∈ (-∞, 1] (higher = better, consistent with :func:`bbox_iou`). The BboxLoss convention
    ``(1 - iou) * weight`` thus yields ``L_WIoU-v3 * weight``.

    Args:
        box1 (torch.Tensor): Predicted boxes, last dim 4.
        box2 (torch.Tensor): Target boxes, last dim 4.
        iou_mean (torch.Tensor): Scalar EMA of mean(L_IoU), maintained by BboxLoss (detached).
        xywh (bool): If True, inputs are (x, y, w, h); else (x1, y1, x2, y2).
        alpha (float): Focusing constant (official default 1.7).
        delta (float): Focusing constant (official default 2.7).
        eps (float): Small value to avoid division by zero.

    Returns:
        (torch.Tensor): WIoU v3 similarity = 1 - L_WIoU-v3.
    """
    # Get coordinates
    if xywh:
        (x1, y1, w1, h1), (x2, y2, w2, h2) = box1.chunk(4, -1), box2.chunk(4, -1)
        w1_, h1_, w2_, h2_ = w1 / 2, h1 / 2, w2 / 2, h2 / 2
        b1_x1, b1_x2, b1_y1, b1_y2 = x1 - w1_, x1 + w1_, y1 - h1_, y1 + h1_
        b2_x1, b2_x2, b2_y1, b2_y2 = x2 - w2_, x2 + w2_, y2 - h2_, y2 + h2_
    else:
        b1_x1, b1_y1, b1_x2, b1_y2 = box1.chunk(4, -1)
        b2_x1, b2_y1, b2_x2, b2_y2 = box2.chunk(4, -1)

    # Intersection
    inter = (b1_x2.minimum(b2_x2) - b1_x1.maximum(b2_x1)).clamp_(0) * (
        b1_y2.minimum(b2_y2) - b1_y1.maximum(b2_y1)
    ).clamp_(0)

    # Union
    w1, h1 = b1_x2 - b1_x1, b1_y2 - b1_y1
    w2, h2 = b2_x2 - b2_x1, b2_y2 - b2_y1
    union = w1 * h1 + w2 * h2 - inter + eps

    # IoU (similarity) and L_IoU (loss form)
    iou = inter / union
    l_iou = 1 - iou

    # Center distance squared (gradient flows through here)
    l2_center = ((b1_x1 + b1_x2 - b2_x1 - b2_x2).pow(2) + (b1_y1 + b1_y2 - b2_y1 - b2_y2).pow(2)) / 4

    # Enclosing box dimensions (detached — no gradient through W_g, H_g)
    cw = b1_x2.maximum(b2_x2) - b1_x1.minimum(b2_x1)
    ch = b1_y2.maximum(b2_y2) - b1_y1.minimum(b2_y1)
    l2_box = cw.pow(2) + ch.pow(2) + eps

    # R_WIoU = exp(center_dist² / (W_g² + H_g²)), ≥ 1
    r_wiou = torch.exp(l2_center / l2_box.detach())

    # L_WIoU-v1 = R_WIoU × L_IoU (multiplication, not addition)
    wiou_v1_loss = r_wiou * l_iou

    # Dynamic focusing coefficient r = β / (δ × α^(β − δ)), fully detached
    with torch.no_grad():
        beta = l_iou.detach() / iou_mean  # outlier degree
        divisor = delta * torch.pow(alpha, beta - delta)
        r = beta / divisor

    # L_WIoU-v3 = r × L_WIoU-v1
    wiou_v3_loss = r * wiou_v1_loss

    # Return similarity (1 - loss), consistent with bbox_iou convention
    return 1 - wiou_v3_loss
```

def _make_forward_inputs(device="cpu"):
    """Helper: construct minimal valid inputs for BboxLoss.forward."""
    n_fg = 10
    pred_dist = torch.randn(n_fg, 4 * 16, requires_grad=True, device=device)
    pred_bboxes = torch.rand(n_fg, 4, requires_grad=True, device=device) * 100
    pred_bboxes[..., 2:] = pred_bboxes[..., :2] + torch.rand(n_fg, 2, device=device) * 50
    anchor_points = torch.rand(8400, 2, device=device) * 640
    target_bboxes = torch.rand(n_fg, 4, device=device) * 100

    target_scores = torch.rand(n_fg, 1, device=device)
    target_scores_sum = torch.tensor(float(n_fg), device=device)
    fg_mask = torch.zeros(8400, dtype=torch.bool, device=device)

    stride = torch.tensor(8.0, device=device)
    return (pred_dist, pred_bboxes, anchor_points, target_bboxes,

class BboxLoss(nn.Module):
    """Criterion class for computing training losses for bounding boxes."""

    def __init__(self, reg_max: int = 16, wiou: bool = False, wiou_alpha: float = 1.0, wiou_momentum: float = 0.01):
        """Initialize the BboxLoss module with regularization maximum and DFL settings."""
        super().__init__()
        self.dfl_loss = DFLoss(reg_max) if reg_max > 1 else None
        self.use_wiou = wiou
        self.wiou_alpha = wiou_alpha
        self.wiou_momentum = wiou_momentum
        # iou_mean EMA buffer: tracks mean(L_IoU) = mean(1 - IoU), initial 1.0 (official default)

"""
YOLO12s + WIoU v3 训练脚本 (E1 实验)
- 预训练权重: yolo12s.pt (COCO)
- 数据集: SSDC-UAV
- 训练: 300 epochs, imgsz=640, batch=16, SGD
- 损失: WIoU v3 替代 CIoU (wiou=True, wiou_alpha=1.0, wiou_momentum=0.01)
"""
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from ultralytics import YOLO


def main():
    # COCO 预训练 YOLO12s
    model = YOLO("yolo12s.pt")

    # SSDC-UAV 数据集配置
    yaml_path = ROOT / "datasets" / "ssdc-uav" / "data.yaml"

    # 训练 (E1: WIoU v3)
    results = model.train(
        data=str(yaml_path),
        epochs=300,
        imgsz=640,
        batch=16,
        optimizer="SGD",
        # WIoU v3 配置
        wiou=True,
        wiou_alpha=1.0,
        wiou_momentum=0.01,
        # 输出
        project="runs/ssdc_uav_train",
        name="yolo12s_WIoUv3_ssdc_uav_exp1",
        device="0",
        save=True,
        # 其他保持默认
        workers=8,

"""
YOLO12s + WIoU v3 评估脚本 (E1 实验)

import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from ultralytics import YOLO


def main():
    # E1 训练好的最佳权重
    weights_path = ROOT / "runs" / "ssdc_uav_train" / "yolo12s_WIoUv3_ssdc_uav_exp1" / "weights" / "best.pt"

        batch=16,
        device="0",

        name="yolo12s_WIoUv3_ssdc_uav_exp1_test",

    main()
